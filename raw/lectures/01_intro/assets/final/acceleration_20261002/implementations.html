01def ddim_quadratic(x, model, betas, steps=50):
02    assert betas.ndim == 1 and betas.numel() >= 2 and steps >= 2
03    dtype = torch.float64 if x.dtype == torch.float64 else torch.float32
04    betas = betas.to(device=x.device, dtype=dtype)
05    T = betas.numel()
06    # Official DDIM quad: equally spaced sqrt(t), then square.
07    roots = np.linspace(0, np.sqrt(0.8 * T), steps)
08    tau = (roots ** 2).astype(int).tolist()  # floor(c * i**2)
09    times = list(reversed(tau)) + [-1]  # -1 means clean data.
10    abar = torch.cat([betas.new_ones(1), (1 - betas).cumprod(0)])
11    def alpha_sigma(t):
12        a = abar[int(t) + 1]
13        return a.sqrt(), (1 - a).sqrt()
14    def eps_fn(xt, t):
15        labels = torch.full((xt.shape[0],), t, device=xt.device)
16        return model(xt, labels)
17    return ddim(x, times, eps_fn, alpha_sigma)

DPM-Solver: Find the midpoint without hiding schedule inversion

01def dpm2_midpoint(t, tn, alpha_sigma):
02    a, s = alpha_sigma(t)
03    an, sn = alpha_sigma(tn)
04    l0, l1 = math.log(a / s), math.log(an / sn)
05    target = 0.5 * (l0 + l1)
06    lo, hi = tn, t
07    # lambda(t)=log(alpha/sigma) decreases as t grows.
08    for _ in range(48):
09        tm = 0.5 * (lo + hi)
10        am, sm = alpha_sigma(tm)
11        if math.log(am / sm) > target:
12            lo = tm
13        else:
14            hi = tm
15    return 0.5 * (lo + hi), l1 - l0

Continuous alpha_sigma; positive alpha/sigma; monotone log-SNR.

t > tn > 0; the midpoint is in log-SNR, not ordinary time.

14def ggdm_moments(p, alpha_noise):

16    a = [p.weights.new_tensor(alpha_noise)]

30def bespoke_reference(velocity, t, ref_times, ref_states):
31    # Linear interpolation for values; ODE derivative for gradients.
32    query = t.detach()
33    hi = torch.searchsorted(ref_times, query).clamp(1, len(ref_times) - 1)
34    lo = hi - 1
35    fraction = (query - ref_times[lo]) / (ref_times[hi] - ref_times[lo])

40def bespoke_error_weights(grid, theta, lipschitz=1.0):
41    h = grid[1:] - grid[:-1]

43    middle = (grid[:-1] + grid[1:]) / 2

01def amed_step(net, head, x, t, u, scale_range=0.01, plugin=False):
02    # net(x, sigma[B]) -> (denoised x0, bottleneck features).
03    batch = x.shape[0]
04    shape = (batch,) + (1,) * (x.ndim - 1)
05    denoised, features = net(x, t.expand(batch))
06    direction = (x - denoised) / t
07    raw = head(features, t.expand(batch), u.expand(batch))
08    ratio = raw[:, 0].sigmoid()
09    scale = 1 + scale_range * (2 * raw[:, 1].sigmoid() - 1)
10    middle = (u.pow(ratio) * t.pow(1 - ratio)).reshape(shape)
11    x_middle = x + (middle - t) * direction
12    d_middle, _ = net(x_middle, middle.reshape(batch))
13    v_middle = (x_middle - d_middle) / middle
14    c = scale.reshape(shape)
15    if plugin:  # Euler to middle, scaled Euler from middle to u.
16        return x_middle + c * (u - middle) * v_middle
17    return x + c * (u - t) * v_middle  # AMED-Solver mean direction

01def ays_times(a, b, u, c=0.5):
02    k = c * c
03    def mass(t):
04        r = k * (b-t)*(b+t) / (t*t * (b*b+k))
05        h = torch.where(r < 1e-3,
06            r*r * (0.5-r/3+r*r/4-r**3/5), r-torch.log1p(r))
07        return h / (2*k*k)
08    z = mass(a)
09    lo, hi = a.expand_as(u), b.expand_as(u)
10    target = u.clamp(1e-9, 1-1e-9) * z
11    for _ in range(48):
12        mid = (lo * hi).sqrt()
13        right = mass(mid) > target
14        lo = torch.where(right, mid, lo)
15        hi = torch.where(right, hi, mid)
16    t = (lo * hi).sqrt()
17    delta = (b-t)*(b+t) / ((t*t+k)*(b*b+k))
18    return t, z, delta

01def pd_coeff(time, x):

03    alpha = torch.cos(angle)
04    alpha = torch.where(time == 1, torch.zeros_like(alpha), alpha)
05    shape = (-1,) + (1,) * (x.ndim - 1)
06    return alpha.reshape(shape), torch.sin(angle).reshape(shape)

01def cd_boundary(raw, x, t, t_min=0.002, sigma_data=0.5):
02    shape = (-1,) + (1,) * (x.ndim - 1)
03    sigma = t.reshape(shape)
04    denom = (sigma.square() + sigma_data**2).sqrt()
05    c_skip = sigma_data**2 / ((sigma - t_min)**2 + sigma_data**2)
06    c_out = sigma_data * (sigma - t_min) / denom
07    c_in = 1 / denom
08    raw_time = 250 * t.log()
09    return c_skip * x + c_out * raw(c_in * x, raw_time)

02    half = dim // 2
03    freq = torch.linspace(0, -math.log(10000), half, device=w.device)

22def add_critic_loss(features, heads, real, fake, gamma=1e-5):
23    freeze(features)
24    heads.requires_grad_(True)
25    with torch.no_grad():
26        real_f, fake_f = features(real.detach()), features(fake.detach())
27    loss = real.new_zeros(())
28    for head, r, f in zip(heads, real_f, fake_f):
29        r = r.detach().requires_grad_(True)
30        rlog = head(r).flatten(1).mean(1)
31        flog = head(f).flatten(1).mean(1)
32        dr = torch.autograd.grad(rlog.sum(), r, create_graph=True)[0]
33        r1 = .5 * dr.flatten(1).square().sum(1).mean()
34        hinge = F.relu(1 - rlog).mean() + F.relu(1 + flog).mean()
35        loss = loss + hinge + gamma * r1
36    return loss

15def dm_gradient(x, teacher, fake_eps, t, noise=None):
16    with torch.no_grad():
17        xt, a, s, _ = vp_noise(x.detach(), t, noise)
18        er, ef = teacher(xt, t), fake_eps(xt, t)
19        # score=-eps/sigma; x0=(xt-sigma*eps)/alpha.
20        real_x0, fake_x0 = (xt - s * er) / a, (xt - s * ef) / a

22        # Appendix B normalization; gradient descent uses fake minus real.
23        grad = (fake_x0 - real_x0) / scale.clamp_min(1e-6)
24    return grad

40    loss.backward()

BOOT: 1. Build the explicit teacher-corrected signal target

Guided BOOT: finite-step signal target, q^-2 weighting, boundary MSE


01@torch.no_grad()
02def boot_target(student, teacher, noise, t, s, cond, uncond, w,
03                alpha_sigma):
04    shape = (-1,) + (1,) * (noise.ndim - 1)
05    at, st = (v.reshape(shape) for v in alpha_sigma(t))
06    a_s, ss = (v.reshape(shape) for v in alpha_sigma(s))
07    yt = student(noise, t, cond)
08    xt = at * yt + st * noise
09    fu, fc = teacher(xt, t, uncond), teacher(xt, t, cond)
10    guided = fu + w * (fc - fu)
11    q = 1 - at * ss / (st * a_s)
12    return yt + q * (guided - yt), q

student(noise, t, cond) predicts the signal y; teacher(x, t, cond) predicts x0.

alpha_sigma is the teacher schedule; 0 < s < t < T avoids singular endpoints.

q = delta * lambda_prime approximation from Algorithm 1; the entire target is stopped.

BOOT: 2. Bootstrap the lower time and anchor the upper boundary

Guided BOOT: finite-step signal target, q^-2 weighting, boundary MSE


13def boot_loss(student, teacher, noise, cond, uncond, alpha_sigma,
14              t_min=0.02, t_max=0.98, delta=0.01, w=7.5, beta=1.0):
15    t = torch.rand(len(noise), device=noise.device)
16    t = t_min + delta + (t_max - t_min - delta) * t
17    s = t - delta
18    target, q = boot_target(student, teacher, noise, t, s,
19                            cond, uncond, w, alpha_sigma)
20    ys = student(noise, s, cond)
21    bootstrap = ((ys - target) / q).square().mean()
22    top = torch.full_like(t, t_max)
23    with torch.no_grad():
24        fu = teacher(noise, top, uncond)
25        fc = teacher(noise, top, cond)
26        boundary_target = fu + w * (fc - fu)
27    boundary = (student(noise, top, cond) - boundary_target).square().mean()
28    return bootstrap + beta * boundary
